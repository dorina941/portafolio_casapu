Java • ejemplo real 
import java.util.*;
import java.util.stream.*;

record Project(String name, String tech, int stars) {}

public class Showcase {
  public static void main(String[] args) {
    List<Project> projects = List.of(
      new Project("Spring API", "Spring Boot", 120),
      new Project("JavaFX Gallery", "JavaFX", 80),
      new Project("Algorithms", "DSA", 55)
    );

    String top = projects.stream()
      .sorted(Comparator.comparingInt(Project::stars).reversed())
      .map(p -> p.name() + " • " + p.tech())
      .collect(Collectors.joining(" | "));

    System.out.println(top);
  }
}
